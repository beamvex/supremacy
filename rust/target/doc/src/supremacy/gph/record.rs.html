1/// One `.GPH` index record (`[si+0]`..`[si+0xA]` in the draw routines).
2#[derive(Debug, Clone, Copy, PartialEq, Eq)]
3pub struct Record {
4    /// Draw position, pixels — `[si+0]`.
5    pub x: u16,
6    /// Draw position, rows — `[si+2]`.
7    pub y: u16,
8    /// Bytes per row for the mode — `ax` on decoder entry, `[si+4]`.
9    pub w: u16,
10    /// Row count — `[si+6]`.
11    pub h: u16,
12    /// Stream offset in paragraphs; the loader adds the heap segment in
13    /// place, so after fixup this is an absolute segment (`ds` at `0x2FB7A`).
14    pub ofs_para: u16,
15    /// `0` = opaque blit, `1` = XOR overlay — `[si+0xA]`.
16    pub mode_flag: u16,
17}
