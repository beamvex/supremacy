1use super::{load_bin, load_index, AssetSet};
2use std::io;
3use std::path::Path;
4
5/// Load one asset set — `dir/STEM.GPH` + `dir/STEM.BIN` — the way the game
6/// does for e.g. `MCG`/`EGAIN`/`CGAWIN`.
7///
8/// # Errors
9/// Propagates the first `fs::read` failure.
10pub fn load_set(dir: &Path, stem: &str, heap_seg: u16) -> io::Result<AssetSet> {
11    let records = load_index(&dir.join(format!("{stem}.GPH")), heap_seg)?;
12    let bin = load_bin(&dir.join(format!("{stem}.BIN")))?;
13    Ok(AssetSet {
14        records,
15        bin,
16        heap_seg,
17    })
18}
