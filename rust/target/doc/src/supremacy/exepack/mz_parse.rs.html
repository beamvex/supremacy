2use super::Error;

4impl Mz {
5    /// Signature + header size `[0x08]` + image length from the page count
6    /// (`[0x02]` bytes-in-last-page, `[0x04]` total pages) + entry `CS`
7    /// `[0x16]`.
8    pub(crate) fn parse(f: &[u8]) -> Result<Self, Error> {
9        let sig = f.get(..2).ok_or(Error::BadMz)?;
10        if sig != b"MZ" && sig != b"ZM" {
11            return Err(Error::BadMz);
12        }
13        let w = |o: usize| -> Result<u16, Error> {
14            let b = f.get(o..o + 2).ok_or(Error::BadMz)?;
15            Ok(u16::from_le_bytes([b[0], b[1]]))
16        };
17        let last = usize::from(w(0x02)?);
18        let pages = usize::from(w(0x04)?);
19        let hdr = usize::from(w(0x08)?) * 16;
20        let img_len = pages * 512 - if last == 0 { 0 } else { 512 - last } - hdr;
21        Ok(Self {
22            image: hdr..hdr + img_len,
23            entry_cs: w(0x16)?,
24        })
25    }
26}
