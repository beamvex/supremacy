7/// Unpack an EXEPACK-compressed MZ executable.
8///
9/// Reproduces the stub's memory model in a flat buffer: `dest` is
10/// `dest_len` paragraphs seeded with the packed load image (the program's
11/// uncompressed head plus the compressed tail), then command blocks expand
12/// the tail backwards from the top. The returned fixup list and real
13/// `cs:ip`/`ss:sp` come from the block header — the values the stub passes
14/// to the far jump at `0x23B76`.
15///
16/// # Errors
17/// `BadMz` (no `MZ`/`ZM` or truncated header), `MissingPack` (no `RB`
18/// header at the entry `CS`), `Corrupt` (bad command byte or a read/write
19/// outside the image).
20pub fn unpack(file: &[u8]) -> Result<Unpacked, Error> {
21    let mz = Mz::parse(file)?;
22    let img = file.get(mz.image.clone()).ok_or(Error::BadMz)?;
23    let blk_ofs = usize::from(mz.entry_cs) * 16;
24    let block = img.get(blk_ofs..).ok_or(Error::MissingPack)?;
25    let hdr = Header::parse(block)?;
26    let mut dest = vec![0u8; usize::from(hdr.dest_len) * 16];
27    let n = dest.len().min(img.len());
28    dest[..n].copy_from_slice(&img[..n]);
29    let si = stream_end(img, blk_ofs)?;
30    let di = dest.len() - 1;
31    run(&mut dest, si, di)?;
32    Ok(Unpacked {
33        relocs: reloc_list(block),
34        image: dest,
35        cs: hdr.real_cs,
36        ip: hdr.real_ip,
37        ss: hdr.real_ss,
38        sp: hdr.real_sp,
39    })
40}
