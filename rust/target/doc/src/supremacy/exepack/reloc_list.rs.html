4/// Parse the packed relocation stream (`0x23B22`-`0x23B44`): 16 batches, one
5/// per `0x1000` paragraphs, each a count word followed by that many `u16`
6/// offsets. Returns the fixup sites as linear offsets into the unpacked
7/// image (`dx * 16 + entry`; the `0xFFFF` special case at `0x23B46` is just
8/// uncanonicalised `seg:ofs` arithmetic).
9///
10/// # Errors
11/// None — a truncated table yields the entries decoded so far.
12#[must_use]
13pub fn reloc_list(block: &[u8]) -> Vec<u32> {
14    let mut out = Vec::new();
15    let mut si = table_ofs(block);
16    for dx in (0u32..=0xF000).step_by(0x1000) {
17        let Ok(n) = word(block, si) else { break };
18        si += 2;
19        for _ in 0..n {
20            if let Ok(ofs) = word(block, si) {
21                out.push(dx * 16 + u32::from(ofs));
22                si += 2;
23            }
24        }
25    }
26    out
27}
