2use super::Unpacked;

4/// 28-byte MZ header for a repackaged image: signature, page geometry,
5/// fixup count/table offset, and the real `ss:sp`/`cs:ip` from the EXEPACK
6/// block. `minalloc`/`maxalloc`/`checksum` follow the unpacked reference.
7pub(super) fn mz_write(u: &Unpacked, hdr_paras: usize) -> Vec<u8> {
8    let total = hdr_paras * 16 + u.image.len();
9    let mut b = Vec::with_capacity(28);
10    b.extend_from_slice(b"MZ");
11    for v in [
12        sat(total % 512),
13        sat(total.div_ceil(512)),
14        sat(u.relocs.len()),
15        sat(hdr_paras),
16        0x74,
17        0xFFFF,
18        u.ss,
19        u.sp,
20        0,
21        u.ip,
22        u.cs,
23        0x1C,
24        0,
25    ] {
26        b.extend_from_slice(&v.to_le_bytes());
27    }
28    b
29}
